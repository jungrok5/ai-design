# 이번 주 워밍업: 두 수의 합
def two_sum(nums, target):
    seen = {}
    for i, x in enumerate(nums):
        if target - x in seen:
            return [seen[target - x], i]
        seen[x] = i

# 해시, 정렬, 투 포인터, BFS/DFS, DP, 그리디 ...
